$('clockOutBtn').onclick=async()=>{ if(!state.active)return; if(state.active.breakOpen){ toast(lang==='fr' ?'Terminez votre pause avant de pointer la sortie.' :'End your break before clocking out.'); return; } // Preserve the original state in case local saving fails. const previousActive=JSON.parse(JSON.stringify(state.active)); const previousRecords=state.records.slice(); const completedRecord={ ...previousActive, breaks:previousActive.breaks.map(b=>({...b})), clockOut:new Date().toISOString() }; // Add the completed shift before clearing the active clock. state.records.push(completedRecord); state.active=null; // Save locally first. const saved=save(); if(!saved){ state.records=previousRecords; state.active=previousActive; render(); toast('SAVE FAILED. Your clock is still active. Download a backup.'); return; } render(); toast(t('clockedOut')); // Upload the completed record immediately when cloud sync is ready. if(cloudSession&&cloudSyncReady&&navigator.onLine){ clearTimeout(cloudSyncTimer); await cloudWriteData(); } };
